EBNF grammar rules for EDL preprocessor

===================================================================================================
directive       : START_MACRO (upload | define) EOL?  // we may handle nested loop.. exit on unexp_symbol (??)
upload          : upload_kw literal symbol
upload_kw       : upload

define          : define_kw symbol (numeric | literal)*
define_kw       : define


operator        : START_MACRO (condition)+ end_operator
condition       : ifdef | ifndef
end_operator    : endif EOL?

ifdef           : ifdef_kw (symbol | logical_expr)+ EOL (directive | operator)* (elifndef | elifdef)* else?
ifndef          : ifndef_kw (symbol | logical_expr)+ EOL (directive | operator)* (elifndef | elifdef)* else?
elifdef         : START_MACRO elifdef_kw (symbol | logical_expr)+ EOL (directive)*                                      // directive always ended with EOL (??)
elifndef        : START_MACRO elifndef_kw (symbol | logical_expr)+ EOL (directive)*
else            : START_MACRO else_kw EOL (directive)*
endif           : START_MACRO endif_kw
logical_expr    : symbol | logical_op (logical_operator (symbol | logical_op)+)+
logical_op      : RP_OP symbol | logical_op (logical_operator (symbol | logical_op)+)+ PR_CL

ifdef_kw        : ifdef
ifndef_kw       : ifndef
elifdef_kw      : elifdef
elifndef_kw     : elifndef
else_kw         : else
endif_kw        : endif
logical_operator: AND | OR

symbol          : [a-zA-Zа-яА-Я_0-9]+
numeric         : [0-9a-fA-FOo\.]+
literal         : [\"\'a-zA-Zа-яА-Я_0-9\/\\]+


START_MACRO     : "#"
EOL             : "\n"
AND             : &&
OR              : ||
